card types
-Placement, may be played on your turn.
-Attack, may be played on your turn.
-Effect, may be played at any time.

--Play area are the white tiles.
--4 Players, use black border.
--3 Players, use grey border.
--Edgewise connected tiles in largest group wins, each tile that is connected at corners is not scored.

1-Start w 4 cards in hand.
2-Roll d20 to go first, turns in clockwise order.
3-One placement and one attack per turn.
4-You must draw draw up to 4 cards at end of your turn. eg, if you use a effect card to prevent an attack, you can only draw up to 4 after your turn is completed.
5-Game ends when no more cards can be drawn from draw pile, last player may still play legal cards.
6-Last turn is triggered when a player cannot make a legal move, everyone else gets one last turn.
7-Can only play on empty tiles unless attack card specifies otherwise.
----------------------------------------------------------------------------------------------------
using UnityEngine;
public class Spawner : MonoBehaviour
{
    private float _nextSpawnTime;
    public int count;
    public int updateFrequency;
    public GameObject square;
    public GameObject circle;
    public GameObject triangle;
    private void FixedUpdate()
    {
        if (!Input.GetKeyDown(KeyCode.Space) || !(Time.time > _nextSpawnTime)) return;
        count++;
        CheckCount();
        _nextSpawnTime = Time.time + updateFrequency;
    }
    
    private void CheckCount()
    {
        switch (count / 10)
        {
            case 0:
                Instantiate(square);
                break;

            case 1:
                Instantiate(circle);
                break;

            case 2:
                Instantiate(triangle);
                break;

            default:
                count = 0;
                break;
        }
    }
}
