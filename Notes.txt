card types
-Placement, may be played on your turn.
-Attack, may be played on your turn.
-Effect, may be played at any time.

--Play area are the white tiles.
--4 Players, use black border.
--3 Players, use grey border.
--Edgewise connected tiles in largest group wins, each tile that is connected at corners is not scored.

1-Start w 4 cards in hand.
2-Roll d20 to go first, turns in clockwise order.
3-One placement and one attack per turn.
4-You must draw draw up to 4 cards at end of your turn. eg, if you use a effect card to prevent an attack, you can only draw up to 4 after your turn is completed.
5-Game ends when no more cards can be drawn from draw pile, last player may still play legal cards.
6-Last turn is triggered when a player cannot make a legal move, everyone else gets one last turn.
7-Can only play on empty tiles unless attack card specifies otherwise.

using UnityEngine;
public class Spawner : MonoBehaviour
{
    private float _nextSpawnTime;
    public int count;
    public int updateFrequency;
    public GameObject square;
    public GameObject circle;
    public GameObject triangle;
    private void FixedUpdate()
    {
        if (!Input.GetKeyDown(KeyCode.Space) || !(Time.time > _nextSpawnTime)) return;
        count++;
        CheckCount();
        _nextSpawnTime = Time.time + updateFrequency;
    }
    
    private void CheckCount()
    {
        switch (count / 10)
        {
            case 0:
                Instantiate(square);
                break;

            case 1:
                Instantiate(circle);
                break;

            case 2:
                Instantiate(triangle);
                break;

            default:
                count = 0;
                break;
        }
    }
}

using System.Collections.Generic;
using UnityEngine;

[CreateAssetMenu(menuName = "Spawn Patterns/Spiral")]
public class SpiralPatternSO : SpawnPatternSO
{
    public float radiusStep = 0.5f;
    public float angleStep = 15f; // degrees between objects

    public override List<Vector3> GetPositions(int count, float spacing)
    {
        var positions = new List<Vector3>();
        float angle = 0f;
        float radius = 0f;

        for (int i = 0; i < count; i++)
        {
            float rad = angle * Mathf.Deg2Rad;
            positions.Add(new Vector3(Mathf.Cos(rad) * radius, 0, Mathf.Sin(rad) * radius));

            angle += angleStep;
            radius += radiusStep * spacing;
        }

        return positions;
    }
}

using System.Collections.Generic;
using UnityEngine;

[CreateAssetMenu(menuName = "Spawn Patterns/Star")]
public class StarPatternSO : SpawnPatternSO
{
    public int points = 5;
    public float radius = 5f;

    public override List<Vector3> GetPositions(int count, float spacing)
    {
        var positions = new List<Vector3>();
        for (int i = 0; i < count; i++)
        {
            int pointIndex = i % points;
            float angle = (pointIndex * 2 * Mathf.PI) / points;
            float r = (i / points) % 2 == 0 ? radius : radius * 0.5f; // alternate long/short
            positions.Add(new Vector3(Mathf.Cos(angle) * r, 0, Mathf.Sin(angle) * r));
        }
        return positions;
    }
}

using System.Collections.Generic;
using UnityEngine;
[CreateAssetMenu(menuName = "Spawn Patterns/Helix")]
public class HelixPatternSO : SpawnPatternSO
{
    public float radius = 5f;
    public float heightStep = 0.5f;

    public override List<Vector3> GetPositions(int count, float spacing)
    {
        var positions = new List<Vector3>();

        for (int i = 0; i < count; i++)
        {
            float angle = i * Mathf.PI * 2 / spacing; // wrap around
            float x = Mathf.Cos(angle) * radius;
            float y = i * heightStep;
            float z = Mathf.Sin(angle) * radius;

            positions.Add(new Vector3(x, y, z));
        }

        return positions;
    }
}

using System.Collections.Generic;
using UnityEngine;
[CreateAssetMenu(menuName = "Spawn Patterns/Cube Grid")]
public class CubeGridPatternSO : SpawnPatternSO
{
    public override List<Vector3> GetPositions(int count, float spacing)
    {
        var positions = new List<Vector3>();
        int cubeSize = Mathf.CeilToInt(Mathf.Pow(count, 1f / 3f)); // cube root

        for (int x = 0; x < cubeSize; x++)
        {
            for (int y = 0; y < cubeSize; y++)
            {
                for (int z = 0; z < cubeSize; z++)
                {
                    positions.Add(new Vector3(x * spacing, y * spacing, z * spacing));
                    if (positions.Count >= count) return positions;
                }
            }
        }
        return positions;
    }
}

using System.Collections.Generic;
using UnityEngine;

[CreateAssetMenu(menuName = "Spawn Patterns/Sphere Surface")]
public class SphereSurfacePatternSO : SpawnPatternSO
{
    public float radius = 5f;

    public override List<Vector3> GetPositions(int count, float spacing)
    {
        var positions = new List<Vector3>();

        // Golden angle method for uniform distribution
        float offset = 2f / count;
        float increment = Mathf.PI * (3f - Mathf.Sqrt(5f)); 

        for (int i = 0; i < count; i++)
        {
            float y = i * offset - 1f + (offset / 2f);
            float r = Mathf.Sqrt(1f - y * y);
            float phi = i * increment;

            float x = Mathf.Cos(phi) * r;
            float z = Mathf.Sin(phi) * r;

            positions.Add(new Vector3(x, y, z) * radius);
        }

        return positions;
    }
}

using System.Collections.Generic;
using UnityEngine;
[CreateAssetMenu(menuName = "Spawn Patterns/Random Height Grid")]
public class RandomHeightGridSO : SpawnPatternSO
{
    public float maxHeight = 5f;

    public override List<Vector3> GetPositions(int count, float spacing)
    {
        var positions = new List<Vector3>();
        int side = Mathf.CeilToInt(Mathf.Sqrt(count));

        for (int x = 0; x < side; x++)
        {
            for (int z = 0; z < side; z++)
            {
                float y = Random.Range(0f, maxHeight);
                positions.Add(new Vector3(x * spacing, y, z * spacing));
                if (positions.Count >= count) return positions;
            }
        }

        return positions;
    }
}

{

        //}
        for (int i = 0; i < length; i++)
        {
            TestGeneratePyramid(i);
        }
        
        
    }

    private void GeneratePyramid(int y)
    {
        int spawnX = length/2;
        int spawnZ = length/2;


        
        
        for (int x = 0; x < length; x++)
        {
            for (int i = 0; i < length; i++)
            {
                Instantiate(prefab, new Vector3(spawnX, y, spawnZ), Quaternion.identity);
                spawnZ--;
            }

            spawnX--;
            spawnZ = length;
        }

        length--;
    }
    
    private void TestGeneratePyramid(int y)
    {
        for (int x = count; x < length - y; x++)
        {
            for (int i = count; i < length - y; i++)
            {
                Instantiate(prefab, new Vector3(i, y, x), Quaternion.identity);
            }
        }

        count++;
        Debug.Log(count);
    }
}
